   119→
   120→export function StorySection({
   121→  source,
   122→  mode,
   123→  onSourceChange,
   124→  onModeChange,
   125→}: {
   126→  source: string;
   127→  mode: M3Mode;
   128→  onSourceChange: (hex: string) => void;
   129→  onModeChange: (m: M3Mode) => void;
   130→}) {
   131→  const trackRef = useRef<HTMLDivElement>(null);
   132→  const [chapter, setChapter] = useState(0);
   133→  const scheme = useMemo(() => buildScheme(source, mode), [source, mode]);
   134→
   135→  const { scrollYProgress } = useScroll({
   136→    target: trackRef,
   137→    offset: ["start start", "end end"],
   138→  });
   139→
   140→  /* Chapter is a pure function of the one scroll authority. */
   141→  useMotionValueEvent(scrollYProgress, "change", (v) => {
   142→    const c = Math.min(
   143→      CHAPTER_COUNT - 1,
   144→      Math.max(0, Math.floor(v * CHAPTER_COUNT))
   145→    );
   146→    setChapter((prev) => (prev === c ? prev : c));
   147→  });
   148→
   149→  const jumpToChapter = useCallback((i: number) => {
   150→    const track = trackRef.current;
   151→    if (!track) return;
   152→    const top = track.offsetTop + (i / CHAPTER_COUNT) * track.clientHeight + 4;
   153→    window.scrollTo({ top, behavior: "auto" });
   154→  }, []);
   155→
   156→  /* Ambient drift — the whole scene breathes against the scroll. */
   157→  const driftY = useTransform(scrollYProgress, [0, 1], [36, -36]);
   158→
   159→  return (
   160→    <section
   161→      ref={trackRef}
   162→      data-testid="story-track"
   163→      aria-label="Zyntrix story"
   164→      style={{ height: `${CHAPTER_COUNT * 100}vh` }}
   165→      className="relative"
   166→    >
   167→      <div className="sticky top-0 z-0 flex h-screen items-center overflow-hidden">
   168→        {/* stage backdrop — theme-derived washes + grain, re-tints live */}
   169→        <StoryBackdrop />
   170→
   171→        {/* scene layer */}
   172→        <motion.div
   173→          style={{ y: driftY }}
   174→          className="relative mx-auto w-full max-w-6xl px-5 pb-28 pt-24 sm:px-10 sm:pb-24 sm:pt-28"
   175→        >
   176→          <AnimatePresence mode="popLayout" initial={false}>
   177→            <motion.div
   178→              key={chapter}
   179→              initial={{ opacity: 0, y: 34, scale: 0.985 }}
   180→              animate={{ opacity: 1, y: 0, scale: 1 }}
   181→              exit={{
   182→                opacity: 0,
   183→                y: -22,
   184→                scale: 0.99,
   185→                transition: { duration: 0.22, ease: EASE },
   186→              }}
   187→              transition={{ duration: 0.5, ease: EASE }}
   188→            >
   189→              {chapter === 0 && <SceneHero source={source} scheme={scheme} />}
   190→              {chapter === 1 && (
   191→                <SceneSeed
   192→                  source={source}
   193→                  scheme={scheme}
   194→                  onSourceChange={onSourceChange}
   195→                />
   196→              )}
   197→              {chapter === 2 && <SceneRoles scheme={scheme} />}
   198→              {chapter === 3 && (
   199→                <SceneDeck
   200→                  source={source}
   201→                  mode={mode}
   202→                  onSourceChange={onSourceChange}
   203→                  onModeChange={onModeChange}
   204→                />
   205→              )}
   206→              {chapter === 4 && <SceneDesktop />}
   207→              {chapter === 5 && <SceneTerminal />}
   208→              {chapter === 6 && <SceneCta />}
   209→            </motion.div>
   210→          </AnimatePresence>
   211→        </motion.div>
   212→
   213→        {/* chapter rail */}
   214→        <nav
   215→          aria-label="Story chapters"
   216→          className="group/rail absolute right-3 top-1/2 hidden -translate-y-1/2 flex-col items-end gap-2.5 sm:right-5 md:flex"
   217→        >
   218→          {COPY.map((c, i) => (
   219→            <ChapterDot
   220→              key={c.no}
   221→              index={i}
   222→              active={chapter === i}
   223→              onJump={() => jumpToChapter(i)}
   224→            />
   225→          ))}
   226→        </nav>
   227→      </div>
   228→    </section>
   229→  );
   230→}
