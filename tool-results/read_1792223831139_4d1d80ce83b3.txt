   337→/** Keep the title bar reachable: allow pushing a window partially out of
   338→ *  the canvas but never further than 92% of its own size. */
   339→function clampGeom(g: Omit<WinGeom, "z">): Omit<WinGeom, "z"> {
   340→  return {
   341→    ...g,
   342→    x: clamp(g.x, -g.w + 0.08, 1 - 0.08),
   343→    y: clamp(g.y, 0, 1 - 0.08),
   344→  };
   345→}

   347→interface DragRef {
   348→  id: WinId;
   349→  pointerId: number;
   350→  startX: number;
   351→  startY: number;
   352→  /** last known-good pointer position — pointercancel events carry 0,0
   353→   *  coords, so commits must never read clientX/Y off the end event */
   354→  lastX: number;
   355→  lastY: number;
   356→  origin: Omit<WinGeom, "z">;
   357→}
   358→
   359→export function DesktopPreview({ mode }: { mode: M3Mode }) {
   360→  // Component is remounted on mode change (keyed by parent), so this only
   361→  // initializes the default wallpaper for the current mode.
   362→  const [wall, setWall] = useState<WallId>(mode === "dark" ? "dark" : "light");
   363→
   364→  const [geoms, setGeoms] = useState<Record<WinId, WinGeom>>(initialGeoms);
   365→  const [draggingId, setDraggingId] = useState<WinId | null>(null);
   366→
   367→  const canvasRef = useRef<HTMLDivElement | null>(null);
   368→  const winEls = useRef<Partial<Record<WinId, HTMLDivElement | null>>>({});
   369→  const dragRef = useRef<DragRef | null>(null);
   370→  const rafRef = useRef(0);
   371→  const zRef = useRef(31);
   372→
   373→  const bringToFront = (id: WinId) => {
   374→    zRef.current += 1;
   375→    const z = zRef.current;
   376→    setGeoms((prev) => (prev[id].z === z ? prev : { ...prev, [id]: { ...prev[id], z } }));
   377→  };
   378→
   379→  const setGeom = (id: WinId, g: Omit<WinGeom, "z">) =>
   380→    setGeoms((prev) => ({ ...prev, [id]: { ...prev[id], ...clampGeom(g) } }));
   381→
   382→  /* ---------------- pointer drag pipeline ---------------- */
   383→
   384→  const beginDrag = (id: WinId, e: React.PointerEvent<HTMLDivElement>) => {
   385→    if (e.button !== 0 && e.pointerType === "mouse") return;
   386→    // Kill text-selection auto-scroll and native drag at the root — this is
   387→    // what previously sent the cursor/selection flying to the top of the page.
   388→    e.preventDefault();
   389→    const winEl = winEls.current[id];
   390→    if (!winEl) return;
   391→    // Canceling pointerdown also suppresses the click-focus, so focus the
   392→    // handle here — preventScroll keeps focus from ever moving the page.
   393→    e.currentTarget.focus({ preventScroll: true });
   394→    try {
   395→      e.currentTarget.setPointerCapture(e.pointerId);
   396→    } catch {
   397→      /* capture is best-effort; handlers still work while the pointer stays */
   398→    }
   399→    dragRef.current = {
   400→      id,
   401→      pointerId: e.pointerId,
   402→      startX: e.clientX,
   403→      startY: e.clientY,
   404→      lastX: e.clientX,
   405→      lastY: e.clientY,
   406→      origin: { x: geoms[id].x, y: geoms[id].y, w: geoms[id].w, h: geoms[id].h },
   407→    };
   408→    setDraggingId(id);
   409→    bringToFront(id);
   410→    document.documentElement.classList.add("window-dragging");
   411→  };
   412→
   413→  const moveDrag = (e: React.PointerEvent<HTMLDivElement>) => {
   414→    const d = dragRef.current;
   415→    if (!d || e.pointerId !== d.pointerId) return;
   416→    d.lastX = e.clientX;
   417→    d.lastY = e.clientY;
   418→    const dx = e.clientX - d.startX;
   419→    const dy = e.clientY - d.startY;
   420→    // Transform-only preview; layout commits once on release. Coalesced to
   421→    // one write per frame so high-frequency events never thrash style.
   422→    if (rafRef.current) cancelAnimationFrame(rafRef.current);
   423→    rafRef.current = requestAnimationFrame(() => {
   424→      rafRef.current = 0;
   425→      const el = winEls.current[d.id];
   426→      if (el) el.style.transform = `translate3d(${dx}px, ${dy}px, 0)`;
   427→    });
   428→  };
   429→
   430→  const endDrag = (e: React.PointerEvent<HTMLDivElement>) => {
   431→    const d = dragRef.current;
   432→    if (!d || e.pointerId !== d.pointerId) return;
   433→    dragRef.current = null;
   434→    if (rafRef.current) {
   435→      cancelAnimationFrame(rafRef.current);
   436→      rafRef.current = 0;
   437→    }
   438→    const el = winEls.current[d.id];
   439→    const canvas = canvasRef.current;
   440→    // Commit from the last known-good pointer position: pointercancel (which
   441→    // can fire with 0,0 coords when e.g. a context menu takes over) must
   442→    // never teleport the window toward the top-left corner.
   443→    const dx = d.lastX - d.startX;
   444→    const dy = d.lastY - d.startY;
   445→    if (el) el.style.transform = "";
   446→    if (canvas && (dx !== 0 || dy !== 0)) {
   447→      const rect = canvas.getBoundingClientRect();
   448→      if (rect.width > 0 && rect.height > 0) {
   449→        setGeom(d.id, {
   450→          x: d.origin.x + dx / rect.width,
   451→          y: d.origin.y + dy / rect.height,
   452→          w: d.origin.w,
   453→          h: d.origin.h,
   454→        });
   455→      }
   456→    }
   457→    setDraggingId(null);
   458→    document.documentElement.classList.remove("window-dragging");
   459→  };
   460→
   461→  useEffect(() => {
   462→    return () => {
   463→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   464→      document.documentElement.classList.remove("window-dragging");
   465→    };
   466→  }, []);
   467→
   468→  /* ---------------- keyboard move (a11y parity) ---------------- */
   469→
   470→  const moveByKeyboard = (id: WinId, e: React.KeyboardEvent<HTMLDivElement>) => {
